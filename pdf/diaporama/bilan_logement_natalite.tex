\subsection{Un modèle à générations imbriquées}


\begin{frame}{Le modèle, acheter avant d'avoir des enfants}

		\node[phase, right=0.75cm of b] (c) {{\color{tdtenavy}\bfseries Retraité}\\[1pt]

	\vspace{0.2cm}
	\raggedright
	\begin{columns}[T]
		\begin{column}{0.315\textwidth}
			\begin{callout}[tdtenavy]{Logement}
				\footnotesize Parc fixe de $H$ m². Acheté jeune, le logement est illiquide jusqu'à la retraite.
			\end{callout}
		\end{column}
		\begin{column}{0.315\textwidth}
			\begin{callout}{Contrainte}
				\footnotesize Le jeune achète sans valoriser les enfants à venir ; s'il les anticipait, la surface serait un \textbf{choix}.
			\end{callout}
		\end{column}
		\begin{column}{0.315\textwidth}
			\begin{callout}[tdtenavy]{Enfants}
				\footnotesize Chaque enfant coûte une part $\phi$ du temps de l'adulte et $\sigma$ m² de son logement.
			\end{callout}
		\end{column}
	\end{columns}

	\vspace{0.15cm}
	\footnotesize Démographie, $N_{t+2} = n_t N_t$. Marché, les jeunes achètent ce que les retraités revendent, et $H = N_t x_t + N_{t-1} x_{t-1}$ à chaque date.
	\source{Article, section 3 ; modèle à générations imbriquées, la cohorte $t$ compte $N_t$ ménages.}
\end{frame}

\begin{frame}{Deux décisions, la surface puis les enfants}
	\begin{columns}[T]
		\begin{column}{0.56\textwidth}
			\footnotesize
			{\color{tdtenavy}\bfseries Jeune adulte}, en $t$
			\[ \max_{c_1,\,x}\ \ln c_1 + \gamma\ln x \quad \text{s.c.} \quad c_1 + p_t x = w_1
			   \;\Longrightarrow\; x_t = \frac{\gamma\, w_1}{(1+\gamma)\, p_t} \]
			{\color{tdtenavy}\bfseries Adulte}, en $t+1$, à surface $x_t$ donnée
			\[ \max_{c_2,\,n}\ \ln c_2 + \gamma\ln(x_t-\sigma n) + \eta\ln n \quad \text{s.c.} \quad c_2 = w_2(1-\phi n) \]
			d'où la condition du premier ordre
			\[ \frac{\eta}{n} = \underbrace{\frac{\phi}{1-\phi n}}_{\text{coût en temps}}
			   + \underbrace{\frac{\gamma\sigma}{x_t-\sigma n}}_{\text{coût en espace}} \]
		\end{column}
		\begin{column}{0.40\textwidth}
			\small
			\begin{itemize}\setlength{\itemsep}{0.45em}
				\item Le jeune consacre une part $\gamma/(1+\gamma)$ de son revenu au logement : sa surface baisse avec le prix qu'il affronte.
				\item L'adulte égalise l'utilité marginale d'un enfant à son coût marginal, en temps et en espace.
				\item Ni le salaire $w_2$ ni le prix courant n'entrent dans la condition : la fécondité ne dépend que de la surface acquise, $n_t = \nu(x_t)$.
			\end{itemize}
		\end{column}
	\end{columns}
	\source{Article, section 3 ; $\gamma$ pondère le logement, $\eta$ les enfants, dans des préférences logarithmiques.}
\end{frame}

\begin{frame}{Fécondité, surface et prix}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\footnotesize
			{\color{tdtenavy}\bfseries Lemme.} $\nu$ est strictement croissante, d'élasticité
			\[ \varepsilon(x) = \frac{x\,\nu'(x)}{\nu(x)} = \frac{E_1+E_2}{E_1+E_2+T_1+T_2} \in (0,1), \]
			où $E_1, E_2$ sont les termes d'espace du coût marginal de l'enfant et $T_1, T_2$ ses termes de temps. Elle décroît avec la surface : un m² compte plus dans un petit logement. {\color{tdtegrey}P1, P5}

			\medskip
			{\color{tdtenavy}\bfseries Proposition, contrainte d'espace.}
			\[ \frac{dn_t}{dp_t} = \nu'(x_t)\,\frac{dx_t}{dp_t} < 0, \qquad \frac{\partial n_t}{\partial p_{t+1}} = 0 \]
			La fécondité dépend du prix payé à l'achat, pas du prix courant ; à surface donnée, le salaire ne joue pas. {\color{tdtegrey}P2 à P4}
		\end{column}
		\begin{column}{0.41\textwidth}
			\begin{callout}[tdtenavy]{Deux régions}
				\footnotesize\RaggedRight Le jeune choisit sa région avec son logement, et l'égalité des utilités donne
				\[ \frac{p_u}{p_r} = \Big(\frac{w_1^u}{w_1^r}\Big)^{\frac{1+\gamma}{\gamma}}, \quad
				   \frac{x_u}{x_r} = \Big(\frac{w_1^u}{w_1^r}\Big)^{-\frac{1}{\gamma}} \]
				Avec $\gamma = 0{,}43$, une prime urbaine de 20\,\% rend les logements urbains plus petits d'un tiers, et la fécondité y est plus faible. À surface donnée, la région ne joue pas. {\color{tdtegrey}P6}
			\end{callout}
		\end{column}
	\end{columns}
	\source{Article, section 3 et annexe A ; $u$ la ville, $r$ la campagne.}
\end{frame}

\begin{frame}{Équilibre, dynamique et ordre de grandeur}
	\begin{columns}[T]
		\begin{column}{0.58\textwidth}
			\footnotesize
			\begin{itemize}\setlength{\itemsep}{0.4em}
				\item \textbf{État stationnaire.} Si la fécondité désirée sans contrainte d'espace, $\bar n = \eta/[\phi(1+\eta)]$, dépasse le remplacement, il est unique :
				\[ x^* = \sigma\Big(1 + \frac{\gamma}{\eta - \phi/(1-\phi)}\Big), \qquad p^* = \frac{\gamma\, w_1}{(1+\gamma)\, x^*} \]
				Une économie plus riche a des logements plus chers, pas plus d'enfants.
				\item \textbf{Dynamique.} Les retraités vendent à leurs enfants, et le parc se transmet dans chaque lignée. Comme $\varepsilon < 1$, la convergence est monotone, sans cycle.
				\item \textbf{Choc sur le parc.} Une variation $\Delta H$ revient tout entière à la cohorte qui achète, $x_{\tau} = x^*(1 + 2\Delta H/H)$ et $n_{\tau} \approx 1 + \varepsilon\, 2\Delta H/H$. À long terme, le nombre de ménages s'ajuste, pas leur surface.
			\end{itemize}
		\end{column}
		\begin{column}{0.38\textwidth}
			\kpi{0,11}{élasticité de la fécondité à la surface à l'état stationnaire, soit +1,1\,\% de fécondité pour 10\,\% de surface}
			\vspace{0.15cm}
			\kpi[tdteorange]{+1,9\,\%}{de fécondité pour la cohorte qui achète si le parc augmente de 10\,\% ; $-$2,8\,\% s'il diminue d'autant}
		\end{column}
	\end{columns}
	\source{Article, section 3 et annexe A. Calibrage : $\gamma = 0{,}43$, $\phi$ de 0,10 à 0,15, $\bar n = 2{,}3/2{,}05$, $x^* = 100$ m², d'où $\sigma$ de 3,4 à 5,7 m².}
\end{frame}

\section{Conclusion}

\begin{frame}{À retenir}
  \small
  \begin{enumerate}\setlength{\itemsep}{0.05em}
    \item Le logement pèse peu dans la baisse de la fécondité, de \textbf{4 à 19\,\%} au plus, et la plupart de l'association est un logement choisi en vue d'un enfant.
    \item Une contrainte existe pourtant, \textbf{à la marge d'une pièce} : sans la place d'un enfant, un tiers de naissances en moins dans les années suivantes, en partie repoussées à plus tard.
    \item Le recensement le confirme de 2014 à 2023, et la situe surtout dans les grandes villes, où la plupart des couples sans enfant n'ont pas cette pièce.
    \item Déménager est plus difficile depuis 2022 (\textbf{crédit, relocation}) ; ce canal pèse au plus 6\,\% du recul des premières naissances, et \textbf{repousse} le premier enfant plus qu'il ne le supprime.
    \item Une politique du logement qui viserait la natalité porterait sur l'\textbf{offre de logements familiaux}, en nombre de pièces.
    \item Le barème de la \textbf{politique familiale} rapporte un quart de moins par enfant qu'en 2014, mais dans l'ERFS ce recul n'a pas d'effet significatif : de 0 à 13\,\% de la baisse selon la période d'estimation, \textbf{au plus un tiers}.
    \item La baisse de la fécondité depuis 2014 tient d'abord au \textbf{recul de la vie en couple} ; la moitié reste inexpliquée.
  \end{enumerate}
\end{frame}
