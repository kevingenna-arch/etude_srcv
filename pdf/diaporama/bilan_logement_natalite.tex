\section{Logement et naissance dans l'enquête SRCV}

\begin{frame}{Les conditions de logement des couples}
  \begin{columns}[T]
    \begin{column}{0.235\textwidth}\kpi{$-$6 m²}{de surface médiane des couples, à moyenne presque inchangée}\end{column}
    \begin{column}{0.235\textwidth}\kpi[tdteorange]{19\,\%}{des couples sans enfant logés sous 50 m², contre 15\,\% en 2014}\end{column}
    \begin{column}{0.235\textwidth}\kpi[tdteorange]{27\,\%}{des couples sans enfant sans place pour un enfant, contre 24\,\%}\end{column}
    \begin{column}{0.235\textwidth}\kpi{$-$6 pts}{de taux d'effort des accédants (de $-$4 à $-$8 selon l'âge)}\end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item Les jeunes locataires ont perdu de la place à taux d'effort stable, jusqu'à $-$8\,\% à 35-39 ans.
    \item Le recul se concentre dans le bas de la distribution et dans les grandes agglomérations.
    \item Les parents à qui une pièce manque passent de 5,6 à 10,5\,\%, ce que le recensement ne confirme pas pour 2019-2023.
  \end{itemize}
  \source{Insee, SRCV 2014 et 2025 ; couples dont la femme a de 15 à 49 ans.}
\end{frame}

\begin{frame}{La surface compte, pas son prix ni le revenu}
  \begin{columns}[T]
    \begin{column}{0.30\textwidth}
      \kpi[tdteorange]{+4\,\%}{de probabilité de naissance pour dix m² de plus (+0,34 point sur un taux de 8,2\,\%)}
      \vspace{0.2cm}
      \kpi{1\,519}{naissances observées dans SRCV, 2014-2025, sur 19\,645 couples}
    \end{column}
    \begin{column}{0.66\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.45em}
        \item \textbf{La surface} est associée à la naissance, davantage pour le premier enfant (+0,59 point).
        \item \textbf{Le coût au mètre carré} n'a aucun effet à surface donnée.
        \item \textbf{Le revenu} non plus, s'il est mesuré avant la naissance : le niveau de vie usuel compte le nouveau-né et les prestations qu'il déclenche.
        \item \textbf{Les accédants} gardent un avantage d'environ 5 points au premier enfant.
        \item Chez les couples arrivés depuis moins de trois ans, l'effet est deux fois et demie plus fort : un premier indice d'anticipation.
      \end{itemize}
    \end{column}
  \end{columns}
  \source{Insee, SRCV 2014-2025 ; probits pondérés, effets marginaux moyens.}
\end{frame}

\begin{frame}{Contrainte ou choix, trois tests d'ordre temporel}
  \begin{columns}[T]
    \begin{column}{0.315\textwidth}
      \card{\faHourglassHalf}{Horizon long}{Sur quatre ans, la surface ne prédit pas les naissances suivantes des couples installés.}
    \end{column}
    \begin{column}{0.315\textwidth}
      \card{\faHome}{Place en pièces}{Sans la place d'un enfant, un tiers de naissances en moins ensuite ($-$35\,\%, $p = 0{,}01$).}
    \end{column}
    \begin{column}{0.315\textwidth}
      \card{\faTruck}{Ordre des événements}{Après un agrandissement, deux fois plus de premières naissances ; l'inverse est rare.}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{callout}{Verdict}
    Une partie de l'association est de l'anticipation, puisqu'on s'agrandit avant l'enfant. L'autre, environ la moitié de l'écart observé en coupe, est une \textbf{contrainte d'indivisibilité}, la pièce qui manque.
  \end{callout}
\end{frame}

\begin{frame}{Le modèle, acheter avant d'avoir des enfants}
  \centering
  \begin{tikzpicture}[
      phase/.style={draw=tdtelight, line width=0.8pt, fill=white, rounded corners=3pt,
        text width=3.75cm, minimum height=1.75cm, align=left, inner sep=5pt, font=\footnotesize},
      quand/.style={font=\scriptsize\bfseries, text=tdteorange, anchor=south west, inner sep=1pt},
      fleche/.style={->, >=stealth, line width=1pt, tdteorange}]
    \node[phase] (a) {{\color{tdtenavy}\bfseries Jeune adulte}\\[1pt]
      Achète $x_t = \gamma w_1/[(1+\gamma)p_t]$ m² au prix $p_t$, pour son usage du moment.};
    \node[phase, right=0.75cm of a] (b) {{\color{tdtenavy}\bfseries Adulte}\\[1pt]
      Choisit $n$ enfants à surface $x_t$ donnée ; chacun coûte $\phi$ de son temps et $\sigma$ m².};
    \node[phase, right=0.75cm of b] (c) {{\color{tdtenavy}\bfseries Retraité}\\[1pt]
      Revend son logement à la cohorte de ses enfants.};
    \node[quand] at (a.north west) {date $t$};
    \node[quand] at (b.north west) {date $t+1$};
    \node[quand] at (c.north west) {date $t+2$};
    \draw[fleche] (a) -- (b);
    \draw[fleche] (b) -- (c);
  \end{tikzpicture}

  \vspace{0.2cm}
  \raggedright
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \footnotesize L'adulte maximise $\ln c_2 + \gamma\ln(x_t-\sigma n) + \eta\ln n$ sous $c_2 = w_2(1-\phi n)$, d'où
      \[ \frac{\eta}{n} = \underbrace{\frac{\phi}{1-\phi n}}_{\text{coût en temps}}
         + \underbrace{\frac{\gamma\sigma}{x_t-\sigma n}}_{\text{coût en espace}} \]
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{callout}{Le cœur du modèle}
        Le jeune adulte achète sans tenir compte des enfants à venir, d'où la \textbf{contrainte}. S'il les anticipait, la surface serait un \textbf{choix}.
      \end{callout}
    \end{column}
  \end{columns}
  \source{Article, section 3 ; générations imbriquées, parc fixe de $H$ m², logement illiquide jusqu'à la retraite.}
\end{frame}

\begin{frame}{Ce que le modèle implique}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{0.35em}
        \item \textbf{La surface élève la fécondité}, avec une élasticité entre 0 et 1, plus forte dans les petits logements. {\color{tdtegrey}P1, P5}
        \item \textbf{Le prix payé à l'achat compte, pas le prix courant.} À surface donnée, ni prix ni salaire ; la surface acquise prédit les naissances à venir. {\color{tdtegrey}P2 à P4}
        \item \textbf{À long terme, la population s'ajuste au parc.} Une économie plus riche a des logements plus chers, pas plus d'enfants.
        \item \textbf{En ville, des logements plus petits.} Avec une prime salariale de 20\,\%, ils sont plus petits d'un tiers, et la fécondité y est plus faible. {\color{tdtegrey}P6}
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \kpi{0,11}{élasticité calibrée de la fécondité à la surface, soit +1,1\,\% de fécondité pour 10\,\% de surface}
      \vspace{0.15cm}
      \kpi[tdteorange]{+1,9\,\%}{de fécondité pour la cohorte qui achète quand le parc s'accroît de 10\,\% (prix $-$17\,\%)}
    \end{column}
  \end{columns}
  \source{Article, section 3. Calibrage : $\gamma = 0{,}43$ (logement à 30\,\% de la dépense), $\phi$ de 0,10 à 0,15, 2,3 enfants souhaités, $x^* = 100$ m², d'où $\sigma$ de 3,4 à 5,7 m².}
\end{frame}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{callout}[tdteok]{Ce qu'il mesure}

      \end{callout}
    \end{column}
  \end{columns}

\begin{frame}{Les anomalies trouvées dans les données}
  \small
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item \textbf{Pondération 2022-2023} faussée pour les ménages avec nouveau-né. Recalée sur les naissances de l'état civil : aucun résultat ne change.
    \item \textbf{Double comptage} : dans les vagues 2022-2025, la moitié des naissances est comptée deux fois ; le niveau de natalité mesuré chute avec la refonte de 2022.
    \item \textbf{Valeurs aberrantes} : surfaces du fichier de panel, 38 mensualités d'emprunt, type de logement qui change de nomenclature sous le même nom, tous corrigés.
    \item \textbf{Code géographique} : absent de la nomenclature en 2014, c'est le Sud-Ouest ; non renseigné en 2010 et 2014, il signale des arrivés récents, sans effet sur les résultats.
    \item \textbf{Date d'emménagement} : la mobilité mesurée par SRCV chute avec la refonte de 2022 ; le recensement sert de référence.
  \end{itemize}
  \vspace{0.1cm}

  \end{callout}
\end{frame}

\begin{frame}{À retenir}
  \small
  \begin{enumerate}\setlength{\itemsep}{0.3em}
    \item Le logement pèse peu dans la baisse de la fécondité, de \textbf{4 à 19\,\%} au plus, et la plupart de l'association est un logement choisi en vue d'un enfant.
    \item Une contrainte existe pourtant, \textbf{à la marge d'une pièce} : sans la place d'un enfant, un tiers de naissances en moins dans les années suivantes.
    \item Le recensement le confirme de 2014 à 2023, et la situe surtout dans les grandes villes, où la plupart des couples sans enfant n'ont pas cette pièce.
    \item Déménager est devenu plus difficile depuis 2022, par le \textbf{crédit et la relocation} plus que par le prix au m² ; ce canal pèse peu dans la baisse des naissances.
    \item Une politique du logement qui viserait la natalité porterait sur l'\textbf{offre de logements familiaux}, en nombre de pièces.
    \item La \textbf{politique familiale} pèserait de 14 à 25\,\% de la baisse, sur les naissances de rang 2 et plus ; l'estimation reste trop imprécise, d'où une étude à part.
    \item La baisse de la fécondité depuis 2014 tient d'abord au \textbf{recul de la vie en couple} ; un tiers reste inexpliqué.
  \end{enumerate}
\end{frame}
